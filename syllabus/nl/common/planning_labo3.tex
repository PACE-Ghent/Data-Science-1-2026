\textbf{Doel:} een dataset inladen en verkennen, rijen en kolommen
selecteren en filteren, nieuwe kolommen toevoegen en de data opkuisen.

\begin{itemize}
  \item \textbf{Opwarmer: waarom niet gewoon Excel?} -- De ruwe tekst van een CSV-bestand bekijken en zien wat een tabelbibliotheek toevoegt.
  \item \textbf{Inladen en verkennen} -- \code{read\_csv}, \code{.head()}, \code{.shape}, \code{.dtypes}, \code{.info()}, \code{.describe()}, \code{mo.ui.table}.
  \item \textbf{Selecteren en filteren} -- Kolommen selecteren, \code{.loc} met logische voorwaarden, voorwaarden combineren, \code{sort\_values}.
  \item \textbf{Nieuwe kolommen} -- Tekst naar getal, leeftijd uit een geboortedatum, snelheid uit een sprinttijd, BMI.
  \item \textbf{Opkuisen} -- Ontbrekende waarden (\code{dropna} of \code{fillna}), tekstlabels, eenheden, dubbels, één volledige opkuis-pipeline.
  \item \textbf{Uitdaging} -- De 5 snelste atleten per positie. Exit ticket.
\end{itemize}

\textbf{Oefeningen:} (1) \code{tests.csv} inladen; (2) aanvallers jonger
dan 16; (3) verdachte sprinttijden; (4) BMI; (5) leeftijd in jaren;
(6) de posities opkuisen; (7) de eenheden van de sprinttijden
corrigeren; (8) de dubbele atleet verwijderen; \textbf{uitdaging:} de
5 snelsten per positie.
